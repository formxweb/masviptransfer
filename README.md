MAS.i18n = (function () {

function pick() {
var saved = null;

}
function t(key, vars) {
var table = MAS.STRINGS[lang] || MAS.STRINGS.en;
var s = table[key];
if (s == null) s = MAS.STRINGS.en[key];
if (s == null) return key;
if (vars) s = s.replace(/\{(\w+)\}/g, function (m, k) { return vars[k] != null ? vars[k] : m; });
return s;
}

if (typeof value === 'string') return value;
return value[lang] || value.en || '';
}

});
}

}

lang = pick();
set(lang, true);
}

function spring(s, target, omega, dt) {

var nx = (x + (s.v + omega * x) * h) * e;

function paint(canvas, name, cap) {

var watched = [];
var io = 'IntersectionObserver' in window ? new IntersectionObserver(function (entries) {
entries.forEach(function (e) {
if (e.isIntersecting) {
var item = e.target._scene;
if (item && !e.target._painted) paint(e.target, item.name, item.cap);
}
});
}, { rootMargin: '600px 0px' }) : null;
function lazy(canvas, name, cap) {
canvas._scene = { name: name, cap: cap };
watched.push(canvas);
if (io) io.observe(canvas); else paint(canvas, name, cap);
}
function refresh() {
watched.forEach(function (c) {

if (!p) return;
var w = c.clientWidth, h = c.clientHeight;

});
